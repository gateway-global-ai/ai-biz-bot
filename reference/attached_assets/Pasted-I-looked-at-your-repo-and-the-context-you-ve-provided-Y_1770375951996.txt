1. The "Business Brain" (Why Your Voice Agent Knows Everything)
This is the magic moment. When they click Voice Concierge, the agent has already ingested:

Google Places data (hours, reviews, photos)
Website content (vectorized)
Uploaded docs (menus, price lists, policies)
Real-time availability (Calendar API)

   Copied // packages/agent-swarm/src/business-brain.ts
import { Pinecone } from '@pinecone-database/pinecone';

export class BusinessBrain {
  constructor(private siteId: string, private vectorStore: Pinecone) {}
  
  async ingestBusinessData(data: BusinessData) {
    // 30-second site build + knowledge base creation
    const chunks = await this.chunkAndVectorize({
      placesData: data.googlePlaces,
      websiteText: data.crawledContent,
      customDocs: data.uploadedPDFs,
      faqs: data.generatedFAQs
    });
    
    await this.vectorStore.upsert({
      namespace: this.siteId,
      vectors: chunks
    });
  }
  
  async getContextForVoiceAgent(query: string, callerHistory?: string[]) {
    // Real-time RAG during voice call
    const results = await this.vectorStore.query({
      namespace: this.siteId,
      topK: 5,
      vector: await this.embed(query)
    });
    
    return {
      businessContext: results.matches,
      tone: this.getBusinessTone(), // From Places data + reviews
      recentCallerContext: callerHistory || [],
      // This makes the agent "know more than employees"
      hiddenKnowledge: results.matches.filter(m => m.score > 0.9)
    };
  }

export class UsageGates {
  async checkCapability(siteId: string, action: Capability) {
    const [tier, usage] = await Promise.all([
      this.getSiteTier(siteId),
      this.getCurrentUsage(siteId)
    ]);
    
    const limits = TIER_LIMITS[tier];
    
    // Check specific features
    if (action === 'telephony' && !limits.telephonyEnabled) {
      throw new UsageError('Upgrade to connect real phone lines', 'TELEPHONY_BLOCKED');
    }
    
    if (action === 'review_filtering' && tier === 'free') {
      throw new UsageError('Filter negative reviews with Pro', 'FEATURE_BLOCKED');
    }
    
    // Check metered usage (voice minutes)
    if (action === 'voice_minute') {
      if (usage.voiceMinutes >= limits.voiceMinutes) {
        await this.sendLimitSMS(siteId);
        throw new UsageError('Voice minutes exhausted', 'LIMIT_REACHED');
      }
      await this.incrementUsage(siteId, 'voiceMinutes');
    }
  }
  
  private async sendLimitSMS(siteId: string) {
    const owner = await db.site.findUnique({ where: { id: siteId }, include: { owner: true }});
    await twilio.messages.create({
      to: owner.phone,
      body: `Your AI Biz Bot is out of voice minutes. 3 customers tried to call. Upgrade to keep answering: https://gatewayglobal.ai/billing`
    });
  }

export class SwarmOrchestrator {
  private agents: Map<string, BaseAgent> = new Map();
  
  constructor(siteId: string) {
    // Load active agents for this business
    this.loadSwarm(siteId);
  }
  
  async handleVoiceCall(callId: string, audioStream: ReadableStream) {
    const context = await businessBrain.getContextForVoiceAgent("");
    
    // Route to appropriate agent based on intent
    const intent = await this.classifyIntent(audioStream);
    
    if (intent === 'booking') {
      return this.agents.get('calendar-coordinator').handle(audioStream, context);
    }
    if (intent === 'complaint') {
      return this.agents.get('escalation-manager').handle(audioStream, context);
    }
    if (intent === 'general_inquiry') {
      return this.agents.get('voice-concierge').handle(audioStream, context);
    }
  }
  
  // For outbound marketing (Pro feature)
  async triggerOutboundCampaign(siteId: string, campaign: Campaign) {
    await usageGates.checkCapability(siteId, 'outbound_calling');
    
    const leads = await this.getLeads(siteId);
    for (const lead of leads) {
      await this.agents.get('outbound-caller').call(lead, {
        script: campaign.script,
        voice: 'cloned-business-owner' // ElevenLabs clone
      });
    }
  }

export async function GET(req: Request) {
  const { searchParams } = new URL(req.url);
  const ref = searchParams.get('ref'); // ?ref=INFLUENCER_CODE
  
  // Set cookie for 60 days
  const response = NextResponse.redirect('/signup');
  response.cookies.set('affiliate_ref', ref, { maxAge: 60 * 24 * 60 * 60 });
  
  return response;
}

model User {
  id            String    @id @default(cuid())
  email         String    @unique
  tier          Tier      @default(FREE)
  sites         Site[]
  affiliateCode String?   @unique
  referrals     AffiliateConversion[]
  stripeCustomerId String?
}

model Site {
  id              String   @id @default(cuid())
  userId          String
  name            String
  subdomain       String   @unique
  customDomain    String?
  
  // The 30-second build data
  businessData    Json?    // Google Places, Yelp, etc.
  vectorNamespace String   // Pinecone namespace
  
  // Feature flags
  telephonyEnabled Boolean @default(false)
  phoneNumber      String?
  googleWorkspaceConnected Boolean @default(false)
  
  // Usage tracking (Redis caches this, DB is source of truth)
  voiceMinutesUsed Int      @default(0)
  lastResetDate    DateTime @default(now())
  
  // Relations
  agents          Agent[]
  usageLogs       UsageLog[]
  affiliate       AffiliateConversion? @relation(fields: [affiliateCode], references: [affiliateCode])
}

model Agent {
  id          String    @id @default(cuid())
  siteId      String
  type        AgentType // VOICE_CONCIERGE, CHAT, OUTBOUND_CALLER, CONTENT_CREATOR
  config      Json      // Personality, voice settings, capabilities
  isActive    Boolean   @default(true)
}

model UsageLog {
  id          String   @id @default(cuid())
  siteId      String
  type        String   // voice_minute, api_call, outbound_call
  quantity    Int
  timestamp   DateTime @default(now())
  cost        Float?   // Calculated cost for internal tracking
}

enum Tier {
  FREE
  PRO
  ENTERPRISE
}

enum AgentType {
  VOICE_CONCIERGE
  CHAT_CONCIERGE
  OUTBOUND_CALLER
  CONTENT_CREATOR
